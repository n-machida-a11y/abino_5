Option Explicit

'================================================================================
' [ファイル] ThisWorkbook（現場用xlsm）
'--------------------------------------------------------------------------------
' 【何のファイル？】
'   現場用xlsm の Excel 開閉時に自動で動く処理。
'   シート保護の再設定や、マクロ専用シート（_snap_系）の表示整理を行う。
'
' 【さわるとき】
'   ほぼ触らない。保護対象シートを増減したい場合のみ Case 文を編集。
'================================================================================


'================================================================================
' ThisWorkbook イベント
' 複数人利用のためのシート保護・フィルタ管理
'================================================================================

' ファイルを開いた時に自動実行
' 全シートのフィルタをクリアし、適切な保護をかけ直す
Private Sub Workbook_Open()
    Dim ws As Worksheet
    Dim originalScreenUpdating As Boolean
    Dim originalDisplayAlerts As Boolean

    originalScreenUpdating = Application.ScreenUpdating
    originalDisplayAlerts = Application.DisplayAlerts
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    On Error GoTo ErrorHandler

    ' 工事番号一覧の B列「NO」(連番) を起動時に必ず非表示化
    '   2026/6/1～ 工事番号で一意管理する仕様変更による
    '   「更新」マクロ未実行でも、ファイルを開いた時点で確実に非表示にする
    On Error Resume Next
    ThisWorkbook.Sheets(SHEET_KOUJI_LIST).Columns("B").Hidden = True
    Err.Clear
    On Error GoTo ErrorHandler

    For Each ws In ThisWorkbook.Worksheets
        ' シートごとに独立してエラー処理（1枚失敗しても他は継続）
        ' interactive:=False でInputBoxを出さないようにする（起動時に邪魔なので）
        On Error Resume Next
        Call SafeUnprotect(ws, interactive:=False)
        Dim unprotectErrNum As Long: unprotectErrNum = Err.Number
        Dim unprotectErrDesc As String: unprotectErrDesc = Err.Description
        Err.Clear

        Call ClearAllFilters(ws)
        Err.Clear
        On Error GoTo ErrorHandler

        ' 保護解除失敗時は警告を出してこのシートはスキップ（処理は継続）
        If unprotectErrNum <> 0 Then
            MsgBox "シート「" & ws.Name & "」の起動時初期化に失敗しました。" & vbCrLf & _
                   "（このシートは以前の保護状態のまま残ります）" & vbCrLf & vbCrLf & _
                   unprotectErrDesc, vbExclamation, "シート保護警告"
            GoTo NextSheet
        End If

        ' シート名に応じて再保護
        Select Case ws.Name
            Case SHEET_IRAISHO
                ' 請求書提出依頼書：完全保護（マクロ専用）
                Call SafeProtectFull(ws)
            Case SHEET_KOUJI_LIST, SHEET_IRAI_RIREKI
                ' 工事番号一覧、依頼履歴：フィルタOK、編集NG
                Call SafeProtectData(ws)
            Case SHEET_KANRI_MASTER, SHEET_OTHER_MASTER
                ' 管理マスタ、その他マスタ：完全保護
                Call SafeProtectFull(ws)
            Case Else
                ' その他のシート（入力フォーム等）は保護しない
        End Select
NextSheet:
    Next ws

Finalize:
    Application.ScreenUpdating = originalScreenUpdating
    Application.DisplayAlerts = originalDisplayAlerts
    Exit Sub

ErrorHandler:
    ' エラーが発生しても起動処理は継続（ユーザー体験優先）
    Resume Finalize
End Sub
